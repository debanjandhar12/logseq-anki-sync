**`api/throttleLogseqRequests.ts`** - refactor to use `async-lock`:
```ts
import AsyncLock from "async-lock";

const lock = new AsyncLock();
const LOGSEQ_REQUEST_INTERVAL_MS = 100;
let nextStartAt = 0;

export function throttleLogseqRequests(logseqApiUrl: string): void {
    const originalFetch = globalThis.fetch;
    if (lock.isBusy?.()) return; // already installed

    globalThis.fetch = async (input, init) => {
        const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url;
        if (url !== logseqApiUrl) return originalFetch(input, init);

        return lock.acquire("logseq-api", async () => {
            const delay = Math.max(0, nextStartAt - Date.now());
            if (delay > 0) await sleep(delay);
            nextStartAt = Date.now() + LOGSEQ_REQUEST_INTERVAL_MS;
            return originalFetch(input, init);
        });
    };
}
```

**`proxy/logseqProxySettings.ts`** - settings management:

**`settings/resetLogseqSettings.ts`** - reset to defaults:

**`settings/persistLogseqSettings.ts`** - persistence:
