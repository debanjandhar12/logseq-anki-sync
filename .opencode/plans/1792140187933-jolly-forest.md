### Key Patterns

1. **Storage Isolation**: Replace `LogseqPluginStorageManager.store` with `InMemoryStore` during tests
2. **Fixture Pattern**: `baseTest.extend()` with async setup/teardown
3. **Driver Pattern**: Page-object-style with `root` locator scoped to mounted container
4. **Modal Mounting**: `mountLauncher(launcher)` returns `{container, result, unmount}`
5. **Confirmation Modals**: Query from `page` (not container) for stacked modals

### C. Editor Readiness Pattern

```typescript
async function waitForEditorReady(driver: EditorDriver) {
    // Wait for dialog to be visible
    await expect.element(driver.dialog).toBeVisible();
    // Wait for loading to complete (loading text should not be present)
    await expect.element(driver.dialog.getByText("Loading")).not.toBeVisible();
    // Wait for sidebar items to be populated or empty state
    const sidebar = driver.sidebar;
    await expect.poll(async () => {
        const items = await sidebar.getByRole("button").count();
        return items >= 0; // 0 is valid (empty state)
    }).toBeDefined();
    // If there's an active file, wait for CodeMirror
    if (await driver.codeMirror.query()) {
        await expect.element(driver.codeMirror.locator(".cm-scroller")).toBeVisible();
    }
}
```

async expectHeaderStyled(header: Locator) {
    const styles = await header.evaluate((el) => {
        const computed = getComputedStyle(el);
        return {
            fontWeight: computed.fontWeight,
            textDecoration: computed.textDecoration
        };
    });
    expect(Number(styles.fontWeight)).toBeGreaterThanOrEqual(600);
    expect(styles.textDecoration).toContain("underline");
}
